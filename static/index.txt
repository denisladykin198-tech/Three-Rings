let AC = null, master = null, muted = localStorage.getItem("tk_sound") === "0";
function audioInit() {
if (AC) return;
try {
AC = new (window.AudioContext || window.webkitAudioContext)();
master = AC.createGain();
master.gain.value = muted ? 0 : .5;
master.connect(AC.destination);
} catch (e) {}
}
function tone(f, dur, type = "sine", vol = .5, delay = 0, slide = 0) {
if (!AC || muted) return;
const t = AC.currentTime + delay;
const o = AC.createOscillator(), gn = AC.createGain();
o.type = type; o.frequency.setValueAtTime(f, t);
if (slide) o.frequency.exponentialRampToValueAtTime(Math.max(30, f + slide), t + dur);
gn.gain.setValueAtTime(0, t); gn.gain.linearRampToValueAtTime(vol, t + .012);
gn.gain.exponentialRampToValueAtTime(.001, t + dur);
o.connect(gn); gn.connect(master); o.start(t); o.stop(t + dur + .05);
}
function noise(dur, vol = .3, delay = 0, hp = 800) {
if (!AC || muted) return;
const t = AC.currentTime + delay;
const len = Math.max(1, (dur * AC.sampleRate) | 0);
const buf = AC.createBuffer(1, len, AC.sampleRate);
const d = buf.getChannelData(0);
for (let i = 0; i < len; i++) d[i] = (Math.random() * 2 - 1) * (1 - i / len);
const src = AC.createBufferSource(); src.buffer = buf;
const f = AC.createBiquadFilter(); f.type = "highpass"; f.frequency.value = hp;
const gn = AC.createGain(); gn.gain.value = vol;
src.connect(f); f.connect(gn); gn.connect(master); src.start(t);
}

else if (to !== "bank") to.cash += amount;
sfx.minus(); renderAll(); cashFlash(p, "down");
if (to !== "bank" && to !== "pot") cashFlash(to, "up");
floatMoney(cellAt(p.ring, p.idx), "−" + fmt(amount), "minus");

};
if (p.cash >= amount) { transfer(); return true; }
log(`<b style="color:${p.color}">${esc(p.name)}</b> не может заплатить ${fmt(amount)} — распродажа!`, "bad");
if (p.bot) {

bankrupt(p, to); return false;

}

const reserve = p => G.cfg.diff === "aggressive" ? 120 : G.cfg.diff === "careful" ? 420 : 260;
function botBuy(p, c) {
if (c.type === "utility") return p.cash - c.price > reserve(p) * .7;
const completes = groupCells(c.group).filter(x => estate(x).owner === p.id).length === groupCells(c.group).length - 1;
return p.cash - c.price > (completes ? reserve(p) * .4 : reserve(p));
}
function botGate(p) {
const own = [0, 0, 0]; ownedCells(p).forEach(x => own[x.ring]++);
let best = 0, bestScore = own[p.ring] * 2 + (p.ring === 0 ? 1 : 0);
for (const d of [1, -1]) {
const r = p.ring + d; if (r < 0 || r > 2) continue;
const sc = own[r] * 2 + (r === 0 && p.cash < 150 ? 2 : 0) + (r === 2 && p.cash > 600 ? 1.5 : 0) + Math.random();
if (sc > bestScore) { bestScore = sc; best = d; }
}
return best;
}

async function botLiquidate(p, amount) {
while (p.cash < amount) {
const withB = ownedCells(p).filter(c => canSellB(p, c));
if (withB.length) { withB.sort((a, b) => a.rents[0] - b.rents[0]); const c = withB[0]; estate(c).level--; p.cash += c.house / 2; continue; }
const sellable = ownedCells(p).filter(c => canSellP(p, c));
if (sellable.length) {
sellable.sort((a, b) => {
const fa = a.type === "prop" && fullGroup(p.id, a.group) ? 1 : 0, fb = b.type === "prop" && fullGroup(p.id, b.group) ? 1 : 0;
return fa - fb || a.price - b.price;
});
const c = sellable[0]; estate(c).owner = null; p.cash += c.price / 2; continue;
}
break;
}
}
async function botBuild(p) {

if (cand.length) {
cand.sort((a, b) => b.rents[0] - a.rents[0]);
const c = cand[0]; estate(c).level++; p.cash -= c.house; sfx.build(); burst(c, ["#3ef08c", "#f5c542"], 16);
log(`<b style="color:${p.color}">${esc(p.name)}</b> строит на «${esc(c.name)}» → ${estate(c).level === 5 ? "ОТЕЛЬ" : "уровень " + estate(c).level}`, "good");
}
}
async function botSell(p) {

if (withB.length) { withB.sort((a, b) => a.rents[0] - b.rents[0]); const c = withB[0]; estate(c).level--; p.cash += c.house / 2; return; }
const sellable = ownedCells(p).filter(c => canSellP(p, c));
if (sellable.length) {
sellable.sort((a, b) => {
const fa = a.type === "prop" && fullGroup(p.id, a.group) ? 1 : 0, fb = b.type === "prop" && fullGroup(p.id, b.group) ? 1 : 0;
return fa - fb || a.price - b.price;
});
const c = sellable[0]; estate(c).owner = null; p.cash += c.price / 2;
log(`<b style="color:${p.color}">${esc(p.name)}</b> продаёт «${esc(c.name)}» банку (+${fmt(c.price / 2)})`);

async function playTurn(p) {
const ep = G.epoch;
try {
if (p.jail > 0) {
const actn = p.bot ? botJail(p) : await humanJailUI(p); g(ep);
if (actn == null) return;
if (actn === "card") { p.jailFree = false; p.jail = 0; log(`<b style="color:${p.color}">${esc(p.name)}</b> использует карту «Освобождение»`, "good"); renderPlayers(); }
else if (actn === "pay") { await pay(p, 50, "bank"); p.jail = 0; log(`<b style="color:${p.color}">${esc(p.name)}</b> выкупается из Изолятора`, "sys"); renderPlayers(); }
else {
const [a, b] = await rollDice(p); g(ep);
if (a === b) {
p.jail = 0; log(`<b style="color:${p.color}">${esc(p.name)}</b> выбрасывает дубль и выходит!`, "good"); renderPlayers();
await moveSteps(p, a + b); await resolveCell(p);
} else {
p.jail++;
if (p.jail >= 3) {
await pay(p, 50, "bank"); p.jail = 0; log(`<b style="color:${p.color}">${esc(p.name)}</b> платит ₽50 после 3 попыток`, "sys"); renderPlayers();
await moveSteps(p, a + b); await resolveCell(p);
} else {
log(`<b style="color:${p.color}">${esc(p.name)}</b> остаётся в Изоляторе (попытка ${p.jail}/3)`, "bad"); renderPlayers();
}
}
}
} else {
let chain = 0;
while (!G.over && !p.bankrupt) {
if (p.bot) { setAction(`<span class="spin"></span> ${esc(p.name)}`, "Бот думает…", []); await S(700); }
else { const hint = chain > 0 ? `Дубль! Бросай ещё (цепочка ${chain})` : ""; await humanRollUI(p, hint); g(ep); }
const [a, b] = await rollDice(p); g(ep);
if (a === b) {
chain++;
if (chain >= 3) { log(`🚨 <b style="color:${p.color}">${esc(p.name)}</b> — три дубля подряд! В Изолятор!`, "bad"); await sendJail(p); break; }
await moveSteps(p, a + b); await resolveCell(p); g(ep); if (G.over || p.bankrupt) break;
} else {
await moveSteps(p, a + b); await resolveCell(p); g(ep); break;
}
}
}
if (!G.over && !p.bankrupt) {
if (p.bot) { await botBuild(p); await S(400); }
else { const a = await humanEndUI(p); g(ep); if (a == null) return; }
}
} catch (e) { if (e !== ABORT) throw e; }
}

        CUR_EPOCH = v.epoch != null ? v.epoch : null;
        G = {
            epoch: 1, players: [], cur: v.cur, turnNo: v.turnNo, pot: v.pot, props: {},
            over: false, winner: null, busy: false, lastSum: v.lastSum,
            decks: { chance: [], treasure: [] }, cfg: { diff: "balanced" }
        };
        buildBoard();
    }
    VIEW_PENDING = v;

    const old = G.players;
    G.players = v.players.map((p, i) => {
        const prev = old[i];
        return {
            id: i, name: p.name || p.botname || BOT_NAMES[i % BOT_NAMES.length],
            color: (p.color || PLAYER_COLORS[i % 4]).trim(), bot: p.bot,
            cash: prev ? prev.cash : p.cash,
            ring: prev ? prev.ring : p.ring,
            idx:  prev ? prev.idx  : p.idx,
            jail: prev ? prev.jail : p.jail,
            jailFree: prev ? prev.jailFree : p.free,
            bankrupt: prev ? prev.bankrupt : p.bankrupt,
            worth: p.worth,
            x: prev ? prev.x : 0, y: prev ? prev.y : 0, tx: 0, ty: 0
        };
    });
    G._wait = v.wait; G._actor = v.actor; G._again = v.again; G._chain = v.chain;
    G._acts = Array.isArray(v.acts) ? v.acts : [];
    G._over = v.over; G._winner = v.winner;

    // новая партия или ресинк — коммитим сразу, без ожидания очереди
    if (newParty || !evPlaying) commitView(newParty);
    renderOnlineActions();
    renderManage();
}

/// Догоняем визуал до серверного состояния. instant=true — при входе/ресинке.
function commitView(instant) {
  const v = VIEW_PENDING; if (!v || !G) return;
  G.cur = v.cur; G.turnNo = v.turnNo; G.pot = v.pot; G.lastSum = v.lastSum;
  G.props = {};
  v.props.forEach(p => {
    const id = flatToId(p.c);
    G.props[id] = { owner: p.o, level: p.l };
  });
  v.players.forEach((p, i) => {
    const q = G.players[i]; if (!q) return;
    q.cash = p.cash; q.ring = p.ring; q.idx = p.idx;
    q.jail = p.jail; q.jailFree = p.free; q.bankrupt = p.bankrupt; q.worth = p.worth;
  });
  layoutTokens(!!instant);
  renderAll();
  G.over = v.over;
  G.winner = v.winner != null ? G.players[v.winner] : null;
  if (v.over && !WIN_SHOWN) { WIN_SHOWN = true; showWin(G.winner); }

}

let evQueue = [], evPlaying = false;
function queueEvent(ev) {
  evQueue.push(ev);
  if (!evPlaying) playNextEv();
}
async function playNextEv() {
    if (evPlaying) return;
    evPlaying = true;
    const ep = G ? G.epoch : 0;
    let aborted = false;
    try {
        while (evQueue.length) {
            const ev = evQueue.shift();
            await playOneEv(ev);
            if (!G || G.epoch !== ep) { aborted = true; break; }
        }
    } catch (e) {
        if (e === ABORT) aborted = true; else console.error("ev", e);
    } finally {
        evPlaying = false;
        if (aborted) { evQueue.length = 0; return; }   // состояние новой партии не трогаем
        commitView(false);
        renderOnlineActions();
        renderManage();
        if (evQueue.length) playNextEv();
    }
}

let NET_WANT = false; 
let NET_JOIN_CMD = null; // Команда для отправки сразу после подключения

function netSend(msg) {
  if (NET.ws && NET.ws.readyState === WebSocket.OPEN) {
    NET.ws.send(msg);
    return true;
  }
  return false;
}

function netConnect(host, joinCmd) {
  NET_WANT = true;
  NET_JOIN_CMD = joinCmd;
  
  // Закрываем старое соединение, если есть
  if (NET.ws) {
    try { NET.ws.close(); } catch(e){}
    NET.ws = null;
  }

  const proto = location.protocol === "https:" ? "wss:" : "ws:";
  const url = `${proto}//${host || location.host}/`;
  
  setRoomUI(NET.code, "wait", "Подключение к серверу...");
  
  try {
    NET.ws = new WebSocket(url);
  } catch (e) {
    toast("Неверный адрес сервера", "err");
    return;
  }

  NET.ws.onopen = () => {
    if (!NET_WANT) return;
    NET.reconnecting = false;
    NET.seq = 0; // Сбрасываем счетчик событий для новой сессии
    
    setRoomUI(NET.code, "live", "Соединение установлено");
    
    // Отправляем команду входа, если она есть (создание или вход по коду/токену)
    if (NET_JOIN_CMD) {
      netSend(NET_JOIN_CMD);
      NET_JOIN_CMD = null;
    } else {
      // Если команды нет, просто запрашиваем список игр (или синхронизируемся, если уже в комнате)
      if (NET.code) {
        netSend("sync");
      } else {
        netSend("games");
      }
    }
  };

  NET.ws.onmessage = e => netHandle(e.data);
  
  NET.ws.onerror = () => {};
  
  NET.ws.onclose = () => {
    if (!NET_WANT) {
      setRoomUI(null, "off", "Нет соединения");
      return;
    }
    
    NET.ws = null;
    setRoomUI(NET.code, "err", "Соединение разорвано");
    
    if (!NET.reconnecting) {
      NET.reconnecting = true;
      toast("Соединение потеряно. Переподключение...", "err");
      
      // Пытаемся переподключиться и восстановить сессию по токену
      setTimeout(() => {
        if (NET_WANT && NET.code && NET.token) {
          // При реконнекте используем токен
          netConnect(host || location.host, `join ${NET.code} ${NET.token}`);
        } else if (NET_WANT) {
          // Если токена нет, просто переподключаемся к хосту
          netConnect(host || location.host, null);
        }
      }, 2000);
    }
  };
}

function netDisconnect() {
    NET_WANT = false; NET_JOIN_CMD = null;
    if (NET.ws) { try { NET.ws.close(); } catch(e){} NET.ws = null; }
    NET.code = null; NET.me = null; NET.token = null; NET.seq = 0;
    localStorage.removeItem("tk_net");
    setRoomUI(null, "off", "Нет соединения");
}

function netHandle(raw) {
  const line = String(raw).trim();
  const sp = line.indexOf(" ");
  const cmd = sp > 0 ? line.slice(0, sp) : line;
  const rest = sp > 0 ? line.slice(sp + 1) : "";

  switch (cmd) {
    case "hello": 
      break;
      
    case "games": 
      break;

    case "room": {
      try { 
        ROOM = JSON.parse(rest); 
        setRoomUI(ROOM.code, "live", `Игроков: ${ROOM.humans}/${ROOM.seats}`);
        renderOnlineActions();
      } catch (e) {}
      break;
    }

    case "joined": {
      const parts = rest.split(/\s+/);
      const code = parts[0];
      const seat = parseInt(parts[1]);
      const token = parts[2];
      
      NET.code = code;
      NET.me = seat;
      NET.token = token;
      
      // Сохраняем токен для реконнекта
      localStorage.setItem("tk_net", JSON.stringify({ code, token }));
      
      MODE = "online";
      $("#setup").classList.remove("on");
      $("#win").classList.remove("on");
      $("#btnNew").disabled = false;
      
      setRoomUI(code, "live", "Вы в комнате");
      toast(`Комната ${code}, место ${seat + 1}`, "gold");
      
      // Запрашиваем полное состояние и инфо о комнате
      netSend("sync");
      netSend("room");
      break;
    }

case "view": {
    const a = rest.indexOf(" ");
    const b = rest.indexOf(" ", a + 1);
    if (a < 0 || b < 0) return;

    const seq = parseInt(rest.slice(0, a), 10);
    const seat = parseInt(rest.slice(a + 1, b), 10);
    const jsonStr = rest.slice(b + 1);

    try {
        const v = JSON.parse(jsonStr);

        // ВАЖНО:
        // Первый view может прийти раньше joined.
        // Поэтому НЕЛЬЗя отбрасывать view только потому, что NET.me ещё null.
        const knownMe = NET.me !== null;
        const forMe = typeof v.you === "number";
        const spectator = seat === 65535 || seat === 4294967295;

        if (knownMe && !forMe && !spectator && seat !== NET.me) {
            return;
        }

        // Если joined ещё не пришёл, но view персональный — берём место из view.
        if (!knownMe && typeof v.you === "number") {
            NET.me = v.you;
        }

        applyView(v);
    } catch (e) {
        console.error("view parse", e, jsonStr.slice(0, 300));
    }

    break;
}
case "ev": {
    const a = rest.indexOf(" ");
    if (a < 0) return;
    try { queueEvent(JSON.parse(rest.slice(a + 1))); }
    catch (e) { console.error("ev parse", e); }
    break;
}

    case "error": {
      toast(rest, "err");
      sfx.err();
      
      // Если ошибка критическая для входа, сбрасываем UI
      if (rest.includes("Ключ не подходит") || rest.includes("Комната не найдена")) {
        netDisconnect();
        $("#setup").classList.add("on");
      } else {
        // При других ошибках пробуем синхронизироваться
        netSend("sync");
      }
      break;
    }

    case "left": {
      toast("Вы вышли из комнаты");
      netDisconnect();
      resetToSetup();
      break;
    }
  }
}

function renderOnlineActions() {
    if (MODE !== "online" || !G) return;
    const w = G._wait || { k: "run" };
    const actor = G._actor;

    if (w.k === "lobby") {
        const iAmHost = ROOM ? ROOM.host === NET.me : NET.me === 0;
        const seats = ROOM ? `${ROOM.humans}/${ROOM.seats}` : "";
        setAction("🏠 Лобби", `Комната ${NET.code}. Игроков: ${seats}.`,
            iAmHost ? [{ label: "▶ Начать партию", cls: "pri", cb: () => netSend("start") }] : []);
        return;
    }
    if (w.k === "over") {
        const iAmHost = ROOM ? ROOM.host === NET.me : NET.me === 0;
        setAction("🏆 Партия окончена", iAmHost ? "Можно пересоздать партию." : "Ждём хоста.",
            iAmHost ? [{ label: "⟳ Новая партия", cls: "pri", cb: () => netSend("restart") }] : []);
        return;
    }
    // Run / нет актёра — сервер занят (бот, анимация, переход хода)
    if (actor == null) { setAction('<span class="spin"></span> Город думает…', "", []); return; }

    const p = G.players[actor];
    if (!p) { setAction("⏳ Ожидание", "", []); return; }
    const isMe = actor === NET.me;
    const dot = `<span class="dot" style="background:${p.color}"></span>`;
    if (!isMe) {
        setAction(dot + " " + esc(p.name),
            p.bot ? '<span class="spin"></span> Бот думает…' : "Ожидание хода другого игрока", []);
        return;
    }
  if (w.k === "over") { setAction("🏆 Партия окончена", "", []); return; }

  // тело-подсказка
  let body = "";
  if (w.k === "buy") {
    const c = cellById(flatToId(w.cell));
    if (c) body = `Купить «${esc(c.name)}» за ${fmt(c.price)}? В кассе ${fmt(p.cash)}.`;
  } else if (w.k === "debt") {
    body = `Кассовый разрыв: нужно ${fmt(w.amount)}. Продавай активы в панели ниже.`;
  } else if (w.k === "jail") {
    body = "Ты в Изоляторе. Выбери способ выхода.";
  } else if (w.k === "gate") {
    body = "Ворота: смени орбиту или останься.";
  } else if (w.k === "roll" && G._again) {
    body = `Дубль! Бросай ещё (цепочка ${G._chain}).`;
  } else if (w.k === "end") {
    body = "Ход завершён. Можно строить — или передавать ход.";
  }

  const styleOf = id => {
    if (id === "roll" || id === "end" || id === "ok" || id === "pay") return "pri";
    if (id === "buy") return "gold";
    if (id === "bankrupt" || id === "concede") return "danger";
    return "ghost";
  };

    const btns = (G._acts || [])
        .filter(a => !["build","sellb","sellp","concede"].includes(a.id))
        .map(a => ({
            label: a.label,
            cls: styleOf(a.id),
            cb: () => {
                let args = (a.arg ?? a.args ?? "").trim();   // <-- ключевая правка
                if (args === "cell") {
                    const flat = w.cell != null ? w.cell : cellFlat(p.ring + ":" + p.idx);
                    args = "cell=" + flat;
                }
                netSend(("act " + a.id + (args ? " " + args : "")).trim());
            }
        }));
    if (!btns.length && (w.k === "card" || w.k === "market"))
        btns.push({ label: "Принять", cls: "pri", cb: () => netSend("act ok") });
    setAction(dot + " " + esc(p.name), body, btns);
}

async function playOneEv(ev) {
    switch (ev.t) {
        case "turn": log(`— Ход ${ev.no}: <b style="color:${G.players[ev.p]?.color || "#fff"}">${esc(G.players[ev.p]?.name || "?")}</b>`, "sys"); break;
        case "dice":
            sfx.dice(); setDie(0, ev.a); setDie(1, ev.b);
            $("#die0").classList.add("landed"); $("#die1").classList.add("landed");
            setTimeout(() => { $("#die0").classList.remove("landed"); $("#die1").classList.remove("landed"); }, 400);
            log(`🎲 <b style="color:${G.players[ev.p]?.color || "#fff"}">${esc(G.players[ev.p]?.name || "?")}</b> выбрасывает <b>${ev.a} + ${ev.b} = ${ev.a + ev.b}</b>${ev.a === ev.b ? " · дубль!" : ""}`, "sys");
            await S(350); break;
        case "walk": {
            const p = G.players[ev.p]; if (!p) break;
            for (let i = 0; i < ev.path.length; i++) {
                const flat = ev.path[i]; const id = flatToId(flat); const c = CELLS.find(x => x.id === id);
                if (c) { p.ring = c.ring; p.idx = c.idx; }
                layoutTokens(); sfx.step();
                if (ev.sal && ev.sal[i] > 0) { sfx.plus(); floatMoney(cellAt(p.ring, p.idx), "+" + fmt(ev.sal[i]), "plus"); }
                await S(ev.back ? 130 : 150);
            }
            if (p) { const c = cellAt(p.ring, p.idx); pulseCell(c); }
            await S(180); break;
        }
        case "teleport": {
            const p = G.players[ev.p]; if (!p) break;
            const c = CELLS.find(x => x.id === flatToId(ev.cell));
            if (c) { p.ring = c.ring; p.idx = c.idx; }
            layoutTokens(); sfx.gate(); await S(480); if (c) pulseCell(c); break;
        }
        case "gain": {
            const p = G.players[ev.p]; if (!p) break;
            sfx.plus(); cashFlash(p, "up");
            const c = CELLS.find(x => x.id === flatToId(ev.cell));
            if (c) floatMoney(c, "+" + fmt(ev.amount), "plus");
            log(`<b style="color:${p.color}">${esc(p.name)}</b> получает ${fmt(ev.amount)}`, "good"); break;
        }
        case "pay": {
            const p = G.players[ev.p]; if (!p) break;
            sfx.minus(); cashFlash(p, "down");
            const toName = ev.to === -1 ? "банку" : ev.to === -2 ? "копилку Привала" : `<b style="color:${G.players[ev.to]?.color || "#fff"}">${esc(G.players[ev.to]?.name || "?")}</b>`;
            const c = CELLS.find(x => x.id === flatToId(ev.cell));
            if (c) floatMoney(c, "−" + fmt(ev.amount), "minus");
            log(`<b style="color:${p.color}">${esc(p.name)}</b> платит ${fmt(ev.amount)} ${toName}`, "bad");
            if (ev.to >= 0 && G.players[ev.to]) cashFlash(G.players[ev.to], "up"); break;
        }
        case "buy": {
            const p = G.players[ev.p]; if (!p) break;
            sfx.buy(); cashFlash(p, "down");
            const c = CELLS.find(x => x.id === flatToId(ev.cell));
            if (c) { burst(c, [c.type === "utility" ? "#ffd23f" : (GROUPS[c.group]?.color || "#fff"), "#ffffff"], 26); floatMoney(c, "−" + fmt(ev.price), "minus"); }
            log(`🏙 <b style="color:${p.color}">${esc(p.name)}</b> покупает «${esc(c?.name || "?")}» за ${fmt(ev.price)}`, "good"); break;
        }
        case "monopoly": {
            const p = G.players[ev.p]; if (!p) break;
            const gr = GROUPS[Object.keys(GROUPS)[ev.group]];
            toast(p.name + " собирает монополию: " + (gr?.name || "?") + " ★", "gold");
            log(`★ <b style="color:${p.color}">${esc(p.name)}</b> собирает всю группу «${gr?.name || "?"}»!`, "good"); break;
        }
        case "pass": {
            const p = G.players[ev.p]; if (!p) break;
            const c = CELLS.find(x => x.id === flatToId(ev.cell));
            log(`<b style="color:${p.color}">${esc(p.name)}</b> отказывается от «${esc(c?.name || "?")}»`);
            if (ev.poor) toast("Не хватает денег на покупку", "err"); break;
        }
        case "rent": {
            const c = CELLS.find(x => x.id === flatToId(ev.cell)); const o = G.players[ev.owner];
            log(`«${esc(c?.name || "?")}» принадлежит <b style="color:${o?.color || "#fff"}">${esc(o?.name || "?")}</b> — аренда ${fmt(ev.amount)}`); break;
        }
        case "card": {
            sfx.card();
            const p = G.players[ev.p]; if (!p) break;
            log(`🃏 <b style="color:${p.color}">${esc(p.name)}</b> тянет карту «${ev.deck === 0 ? "Шанс" : "Казна"}»`, "card");
            const deck = ev.deck === 0 ? "chance" : "treasure";
            const card = { e: ev.e, t: ev.text };
            if (p.id === NET.me) {
                // только показываем; подтверждающая кнопка придёт из view.wait = card
                const m = openModal(false);
                m.innerHTML = cardHtml(deck, card) +
                  `<div class="mrow"><button class="btn pri big" id="cardOk">Принять судьбу</button></div>`;
                await new Promise(res => {
                  const done = () => { closeModal(); res(); };
                  m.querySelector("#cardOk").addEventListener("click", () => { sfx.click(); done(); });
                  CARD_OK = done;                       // вызовется из renderOnlineActions/netHandle
                });
                CARD_OK = null;
                netSend("act ok");
            } else {
                const m = openModal(false);
                m.innerHTML = cardHtml(deck, card);
                await S(actorIsBot(p) ? 1100 : 6000); 
                closeModal();
            }
            break;
        }
        case "market": {
            sfx.card();
            const p = G.players[ev.p]; if (!p) break;
            log(`📈 Биржа: ${ev.name} (${sign(ev.delta)}) для <b style="color:${p.color}">${esc(p.name)}</b>`, ev.delta > 0 ? "good" : "bad");

            const p = G.players[ev.p]; if (!p) break;

        }

        }

            const p = G.players[ev.p]; if (!p) break;
            sfx.gate(); const c = CELLS.find(x => x.id === flatToId(ev.cell));
            if (c) { burst(c, ["#27e0ff", "#5aa2ff"], 18); pulseCell(c); }

        }
        case "potempty": { toast("Привал пуст — копилка ещё не наполнилась"); log("Привал пуст"); break; }
        case "bankrupt": {
            const p = G.players[ev.p]; if (!p) break;
            sfx.jail(); toast(p.name + " выбывает из игры", "err");

            log(`💥 <b style="color:${p.color}">${esc(p.name)}</b> — банкрот! Имущество ${toName}`, "bad"); break;
        }
        case "win": {
            sfx.win(); confetti(); setTimeout(confetti, 700);

        }
        case "debt": {

        }
        case "begin": log("🏙 Партия началась! Удачи, магнаты.", "sys"); break;
        case "create": log(`Комната создана. Игра: ${ev.game}, мест: ${ev.seats}`, "sys"); break;
        case "join": {
            const q = G && G.players[ev.seat];

            break;
        }
        case "reset":
		    hardResetVisual();

		    break;
        case "leave": {
            log(`Место ${ev.seat + 1} освободилось`, "sys");
            netSend("room");
            break;
        }
        case "boton": {
            const q = G && G.players[ev.p];
            log(`<b style="color:${q?.color || "#fff"}">${esc(q?.name || "?")}</b> уходит — место занимает бот`, "sys");
            break;
        }
        case "botoff": {
            const q = G && G.players[ev.p];
            log(`<b style="color:${q?.color || "#fff"}">${esc(q?.name || "?")}</b> снова за пультом`, "good");
            break;
        }
    }
}

function flatToId(flat) {
    let acc = 0;
    for (let r = 0; r < 3; r++) {
        const len = 4 * RING_META[r].k + 4;
        if (flat < acc + len) return r + ":" + (flat - acc);
        acc += len;
    }
    return "0:0";
}
